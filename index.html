const SHEET_NAME_SALES = "Sales"; const EXPENSE_SHEET = "Expenses"; const PRICE_SHEET = "Prices"; const IMAGE_SHEET = "Images"; // เผื่อไว้เก็บรูปภาพตามต้องการ function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sales = []; var salesSheet = ss.getSheetByName(SHEET_NAME_SALES); if (salesSheet) { var data = salesSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0]) { var dateVal = String(data[i][1] || '').trim(); // 🛡️ ระบบกรอง: ข้ามแถวข้อมูลที่วันที่รวน หรือข้อมูลหลุดตำแหน่ง if (!dateVal.includes('-') || dateVal.includes('178749') || dateVal.length > 15) { continue; } var up = Number(data[i][4]) || 0; var q = Number(data[i][5]) || 1; // ป้องกันกรณีสลับช่องราคาและจำนวนในข้อมูลเก่า if (up > 50 && q <= 5 && up % 5 !== 0 && q > 1) { var temp = up; up = q; q = temp; } sales.push({ id: String(data[i][0]), ts: Number(data[i][1]) || Date.now(), date: dateVal, name: String(data[i][2] || ''), unitPrice: up, qty: q, totalPrice: up * q, category: String(data[i][7] || 'เครื่องดื่ม'), paymentMethod: String(data[i][8] || 'cash'), isDelivery: String(data[i][9]) === 'true', orderType: String(data[i][10] || 'หน้าร้าน'), custName: String(data[i][11] || ''), custPhone: String(data[i][12] || ''), custAddress: String(data[i][13] || ''), custGps: String(data[i][14] || ''), time: String(data[i][15] || '12:00') }); } } } // ดึงข้อมูลต้นทุน var expenses = []; var expSheet = ss.getSheetByName(EXPENSE_SHEET); if (expSheet) { var expData = expSheet.getDataRange().getValues(); for (var j = 1; j < expData.length; j++) { if (expData[j][0]) { expenses.push({ id: String(expData[j][0]), ts: Number(expData[j][1]) || 0, date: String(expData[j][2]), name: String(expData[j][3]), cost: Number(expData[j][4]) || 0, time: String(expData[j][5]) }); } } } // ดึงข้อมูลราคา var prices = {}; var priceSheet = ss.getSheetByName(PRICE_SHEET); if (priceSheet) { var pData = priceSheet.getDataRange().getValues(); for (var k = 1; k < pData.length; k++) { if (pData[k][0]) { prices[String(pData[k][0])] = Number(pData[k][1]) || 0; } } } // ดึงข้อมูลรูปภาพ var images = {}; var imageSheet = ss.getSheetByName(IMAGE_SHEET); if (imageSheet) { var iData = imageSheet.getDataRange().getValues(); for (var m = 1; m < iData.length; m++) { if (iData[m][0]) { images[String(iData[m][0])] = String(iData[m][1]); } } } var result = { sales: sales, expenses: expenses, prices: prices, images: images }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.action === "add_sale") { var sheet = getOrCreateSheet(ss, SHEET_NAME_SALES, ["ID", "Date", "Name", "UnitPrice", "Qty", "TotalPrice", "Category", "PaymentMethod", "OrderType", "CustName", "CustPhone", "CustAddress", "CustGps", "Time"]); var item = data.item; sheet.appendRow([ item.id, item.date, item.name, item.unitPrice, item.qty, item.totalPrice, item.category, item.paymentMethod, item.orderType, item.custName, item.custPhone, item.custAddress, item.custGps, item.time ]); } else if (data.action === "delete_sale") { deleteRowById(ss.getSheetByName(SHEET_NAME_SALES), data.id); } else if (data.action === "update_sale") { // อัปเดตสถานะหรือรายการ updateOrAppendRow(ss.getSheetByName(SHEET_NAME_SALES), data.item); } else if (data.action === "add_expense") { var sheet = getOrCreateSheet(ss, EXPENSE_SHEET, ["ID", "Timestamp", "Date", "Name", "Cost", "Time"]); var item = data.item; sheet.appendRow([item.id, item.ts, item.date, item.name, item.cost, item.time]); } else if (data.action === "delete_expense") { deleteRowById(ss.getSheetByName(EXPENSE_SHEET), data.id); } else if (data.action === "update_price") { setKeyValue(ss.getSheetByName(PRICE_SHEET), data.name, data.price); if(data.imageUrl) { setKeyValue(ss.getSheetByName(IMAGE_SHEET), data.name, data.imageUrl); } } else if (data.action === "delete_price") { removeKey(ss.getSheetByName(PRICE_SHEET), data.name); removeKey(ss.getSheetByName(IMAGE_SHEET), data.name); } return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateSheet(ss, name, headers) { var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headers && headers.length > 0) { sheet.appendRow(headers); } } return sheet; } function deleteRowById(sheet, id) { if (!sheet) return; var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; } } } function updateOrAppendRow(sheet, item) { if (!sheet) return; var rows = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(item.id)) { sheet.getRange(i + 1, 8).setValue(item.paymentMethod); // อัปเดตประเภทการชำระเงิน found = true; break; } } if (!found) { sheet.appendRow([ item.id, item.date, item.name, item.unitPrice, item.qty, item.totalPrice, item.category, item.paymentMethod, item.orderType, item.custName, item.custPhone, item.custAddress, item.custGps, item.time ]); } } function setKeyValue(sheet, key, val) { if (!sheet) return; var rows = sheet.getDataRange().getValues(); var found = false; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(key)) { sheet.getRange(i + 1, 2).setValue(val); found = true; break; } } if (!found) { sheet.appendRow([key, val]); } } function removeKey(sheet, key) { if (!sheet) return; var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === String(key)) { sheet.deleteRow(i + 1); break; } } }
