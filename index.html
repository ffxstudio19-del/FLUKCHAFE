function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. ดึงข้อมูลยอดขาย (Sales) var salesSheet = getOrCreateSheet(ss, "Sales", [ "ID", "Timestamp", "Date", "Time", "Menu Name", "Category", "Payment Method", "Unit Price", "Qty", "Total Price" ]); var salesData = []; var salesRows = salesSheet.getDataRange().getValues(); if (salesRows.length > 1) { for (var i = 1; i < salesRows.length; i++) { var r = salesRows[i]; if (r[0]) { salesData.push({ id: String(r[0]), ts: Number(r[1]) || 0, date: formatDateString(r[2]), time: String(r[3]), name: String(r[4]), category: String(r[5]), paymentMethod: String(r[6]) || "cash", unitPrice: Number(r[7]) || 0, qty: Number(r[8]) || 1, totalPrice: Number(r[9]) || (Number(r[7]) * Number(r[8])) }); } } } // 2. ดึงข้อมูลต้นทุน (Expenses) var expSheet = getOrCreateSheet(ss, "Expenses", [ "ID", "Timestamp", "Date", "Time", "Item Name", "Cost" ]); var expData = []; var expRows = expSheet.getDataRange().getValues(); if (expRows.length > 1) { for (var j = 1; j < expRows.length; j++) { var er = expRows[j]; if (er[0]) { expData.push({ id: String(er[0]), ts: Number(er[1]) || 0, date: formatDateString(er[2]), time: String(er[3]), name: String(er[4]), cost: Number(er[5]) || 0 }); } } } // 3. ดึงข้อมูลราคาเมนู (Prices) var priceSheet = getOrCreateSheet(ss, "Prices", ["Menu Name", "Price"]); var priceData = {}; var priceRows = priceSheet.getDataRange().getValues(); if (priceRows.length > 1) { for (var k = 1; k < priceRows.length; k++) { var pr = priceRows[k]; if (pr[0]) { priceData[String(pr[0])] = Number(pr[1]); } } } var response = { status: "success", sales: salesData, expenses: expData, prices: priceData }; return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); // ป้องกันข้อมูลชนกันเวลากดพร้อมกันหลายเครื่อง try { var rawData = e.postData.contents; var data = JSON.parse(rawData); var action = data.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // --- จัดการยอดขาย (Sales) --- if (action === "add_sale") { var sheet = getOrCreateSheet(ss, "Sales"); var item = data.item; sheet.appendRow([ String(item.id), item.ts || Date.now(), item.date, item.time, item.name, item.category, item.paymentMethod || "cash", item.unitPrice, item.qty, item.totalPrice ]); } else if (action === "delete_sale") { var sheet = getOrCreateSheet(ss, "Sales"); deleteRowById(sheet, String(data.id)); } else if (action === "update_sale") { var sheet = getOrCreateSheet(ss, "Sales"); updateSaleRow(sheet, data.item); } // --- จัดการต้นทุน (Expenses) --- else if (action === "add_expense") { var sheet = getOrCreateSheet(ss, "Expenses"); var item = data.item; sheet.appendRow([ String(item.id), item.ts || Date.now(), item.date, item.time, item.name, item.cost ]); } else if (action === "delete_expense") { var sheet = getOrCreateSheet(ss, "Expenses"); deleteRowById(sheet, String(data.id)); } // --- จัดการราคาเมนู (Prices) --- else if (action === "update_price") { var sheet = getOrCreateSheet(ss, "Prices"); updateOrAddPrice(sheet, data.name, data.price); } else if (action === "delete_price") { var sheet = getOrCreateSheet(ss, "Prices"); deletePriceRow(sheet, data.name); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // ---------------- Helper Functions ---------------- function getOrCreateSheet(ss, name, headers) { var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (headers && headers.length > 0) { sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e2e8f0"); } } return sheet; } function deleteRowById(sheet, id) { var values = sheet.getDataRange().getValues(); for (var i = values.length - 1; i >= 1; i--) { if (String(values[i][0]) === id) { sheet.deleteRow(i + 1); break; } } } function updateSaleRow(sheet, item) { var values = sheet.getDataRange().getValues(); var targetId = String(item.id); for (var i = 1; i < values.length; i++) { if (String(values[i][0]) === targetId) { // อัปเดตข้อมูลแถวที่ตรงกัน (รวมถึงวิธีจ่ายเงิน Cash/PromptPay) sheet.getRange(i + 1, 7).setValue(item.paymentMethod || "cash"); break; } } } function updateOrAddPrice(sheet, name, price) { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (String(values[i][0]).trim().toLowerCase() === String(name).trim().toLowerCase()) { sheet.getRange(i + 1, 2).setValue(price); return; } } sheet.appendRow([name, price]); } function deletePriceRow(sheet, name) { var values = sheet.getDataRange().getValues(); for (var i = values.length - 1; i >= 1; i--) { if (String(values[i][0]).trim().toLowerCase() === String(name).trim().toLowerCase()) { sheet.deleteRow(i + 1); break; } } } function formatDateString(val) { if (val instanceof Date) { var y = val.getFullYear(); var m = String(val.getMonth() + 1).padStart(2, '0'); var d = String(val.getDate()).padStart(2, '0'); return y + "-" + m + "-" + d; } return String(val); }
