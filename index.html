function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var salesSheet = getSalesSheet(ss); var expSheet = getOrCreateSheet(ss, "Expenses", ["Date", "Time", "Expense Name", "Cost", "Expense ID"]); var priceSheet = getOrCreateSheet(ss, "Prices", ["Menu Name", "Price", "Recipe"]); var action = (e && e.parameter && e.parameter.action) || (e && e.postData && JSON.parse(e.postData.contents).action) || "get_all"; var payload = (e && e.postData && e.postData.contents) ? JSON.parse(e.postData.contents) : (e ? e.parameter : {}); if (action === "get_all") { var salesData = readSalesSheetData(salesSheet); var expData = readExpSheetData(expSheet); var priceData = readPricesData(priceSheet); return jsonResponse({ status: "success", sales: salesData, expenses: expData, prices: priceData.prices, recipes: priceData.recipes }); } if (action === "add_sale") { var items = payload.items || [payload]; var existingIds = getExistingIds(salesSheet); items.forEach(function(it) { var itId = String(it.id || ("ORD-" + Date.now())); if (existingIds.indexOf(itId) !== -1) return; var dateStr = String(it.date || Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd")); var timeStr = String(it.time || Utilities.formatDate(new Date(), "GMT+7", "HH:mm:ss")); var unitPrice = Number(it.unitPrice) || 0; var qty = Number(it.qty) || 1; var totalPrice = Number(it.totalPrice) || (unitPrice * qty); var status = it.status || "pending"; var statusText = totalPrice + " " + status; var nextRow = salesSheet.getLastRow() + 1; // บันทึกลงล็อกแต่ละคอลัมน์ให้ตรงเป๊ะ (Col A ถึง K) salesSheet.getRange(nextRow, 1, 1, 11).setValues([[ itId, // Col A: ID itId, // Col B: Order ID dateStr, // Col C: Date timeStr, // Col D: Time it.name, // Col E: Menu Name it.category || "เครื่องดื่ม", // Col F: Category it.paymentMethod || "cash", // Col G: Payment unitPrice, // Col H: Price qty, // Col I: Qty it.timestamp || Date.now(), // Col J: Timestamp statusText // Col K: Total + Status ]]); // บังคับเซ็ตคอลัมน์ Date และ Time เป็น Text ธรรมดา salesSheet.getRange(nextRow, 3).setNumberFormat("@"); salesSheet.getRange(nextRow, 4).setNumberFormat("@"); }); return jsonResponse({ status: "success" }); } if (action === "update_status") { var r = findRowById(salesSheet, String(payload.id)); if (r > 0) { var curK = String(salesSheet.getRange(r, 11).getValue() || ""); var numMatch = curK.match(/^([\d.]+)/); var pricePart = numMatch ? numMatch[1] : ""; salesSheet.getRange(r, 11).setValue((pricePart + " " + payload.status).trim()); } return jsonResponse({ status: "success" }); } if (action === "update_payment") { var r = findRowById(salesSheet, String(payload.id)); if (r > 0) { salesSheet.getRange(r, 7).setValue(payload.paymentMethod); } return jsonResponse({ status: "success" }); } if (action === "delete_sale") { var r = findRowById(salesSheet, String(payload.id)); if (r > 0) salesSheet.deleteRow(r); return jsonResponse({ status: "success" }); } if (action === "edit_sale") { var r = findRowById(salesSheet, String(payload.id)); if (r > 0) { salesSheet.getRange(r, 5).setValue(payload.name); salesSheet.getRange(r, 6).setValue(payload.category); salesSheet.getRange(r, 8).setValue(payload.unitPrice); salesSheet.getRange(r, 9).setValue(payload.qty); salesSheet.getRange(r, 11).setValue(payload.totalPrice + " done"); } return jsonResponse({ status: "success" }); } if (action === "add_expense") { expSheet.appendRow([ payload.date || Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd"), payload.time || Utilities.formatDate(new Date(), "GMT+7", "HH:mm:ss"), payload.name, payload.cost || 0, payload.id || ("EXP-" + Date.now()) ]); return jsonResponse({ status: "success" }); } if (action === "delete_expense") { var data = expSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][4]) === String(payload.id)) { expSheet.deleteRow(i + 1); break; } } return jsonResponse({ status: "success" }); } if (action === "update_price") { var pData = priceSheet.getDataRange().getValues(); var found = -1; for (var i = 1; i < pData.length; i++) { if (String(pData[i][0]).trim() === String(payload.name).trim()) { found = i + 1; break; } } if (found > 0) { priceSheet.getRange(found, 2).setValue(payload.price); if (payload.recipe !== undefined) priceSheet.getRange(found, 3).setValue(payload.recipe); } else { priceSheet.appendRow([payload.name.trim(), payload.price, payload.recipe || ""]); } return jsonResponse({ status: "success" }); } if (action === "delete_price") { var pData = priceSheet.getDataRange().getValues(); for (var i = 1; i < pData.length; i++) { if (String(pData[i][0]).trim() === String(payload.name).trim()) { priceSheet.deleteRow(i + 1); break; } } return jsonResponse({ status: "success" }); } return jsonResponse({ status: "error", message: "Unknown action" }); } catch (err) { return jsonResponse({ status: "error", message: err.toString() }); } finally { lock.releaseLock(); } } function getSalesSheet(ss) { var s = ss.getSheetByName("Sales"); if (s && s.getLastRow() > 1) return s; return ss.getSheets()[0]; } function getOrCreateSheet(ss, name, headers) { var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#f1f5f9"); } return sheet; } function readSalesSheetData(sheet) { var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var rows = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (!row[4] && !row[2]) continue; var id = String(row[0] || ("old-" + i)); var dateRaw = row[2]; var timeRaw = row[3]; var name = String(row[4] || "").trim(); if (!name) continue; var category = String(row[5] || "เครื่องดื่ม").trim(); var payMethod = String(row[6] || "cash").trim(); var unitPrice = Number(row[7]) || 0; var qty = Number(row[8]) || 1; var timestampCol = Number(row[9]) || 0; var colK = String(row[10] || "").trim(); var dateStr = ""; if (dateRaw instanceof Date) { dateStr = Utilities.formatDate(dateRaw, "GMT+7", "yyyy-MM-dd"); } else { dateStr = String(dateRaw || "").trim().replace(/^['"]/, ''); } var timeStr = ""; if (timeRaw instanceof Date) { timeStr = Utilities.formatDate(timeRaw, "GMT+7", "HH:mm:ss"); } else { timeStr = String(timeRaw || "12:00:00").trim().replace(/^['"]/, ''); if (timeStr.length === 5) timeStr += ":00"; } var timestamp = timestampCol > 0 ? timestampCol : new Date(dateStr + "T" + timeStr + "+07:00").getTime(); if (isNaN(timestamp)) timestamp = Date.now(); var status = colK.toLowerCase().indexOf("pending") !== -1 ? "pending" : "done"; // ตรวจสอบราคารวมใน Col K ถ้าว่างหรือคำนวณไม่ได้ ให้ใช้ unitPrice * qty แทนเพื่อแก้ปัญหายอดรวมขึ้น 0 บาท var numMatch = colK.match(/^([\d.]+)/); var totalPrice = numMatch ? Number(numMatch[1]) : 0; if (!totalPrice || totalPrice <= 0) { totalPrice = unitPrice * qty; } rows.push({ id: id, timestamp: timestamp, date: dateStr, time: timeStr, name: name, category: category, paymentMethod: payMethod, unitPrice: unitPrice, qty: qty, totalPrice: totalPrice, status: status }); } return rows; } function readExpSheetData(sheet) { var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var rows = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (!row[0] && !row[2]) continue; var dateStr = row[0] instanceof Date ? Utilities.formatDate(row[0], "GMT+7", "yyyy-MM-dd") : String(row[0]||"").replace(/^['"]/, ''); var timeStr = row[1] instanceof Date ? Utilities.formatDate(row[1], "GMT+7", "HH:mm:ss") : String(row[1]||"").replace(/^['"]/, ''); var parsedDt = new Date(dateStr + "T" + (timeStr.length === 5 ? timeStr + ":00" : (timeStr || "12:00:00"))); rows.push({ timestamp: isNaN(parsedDt.getTime()) ? Date.now() : parsedDt.getTime(), date: dateStr, time: timeStr, name: String(row[2]||""), cost: Number(row[3])||0, id: String(row[4]||"") }); } return rows; } function readPricesData(sheet) { var data = sheet.getDataRange().getValues(); var prices = {}, recipes = {}; for (var i = 1; i < data.length; i++) { var name = String(data[i][0]).trim(); if (name) { prices[name] = Number(data[i][1]) || 0; recipes[name] = String(data[i][2] || ""); } } return { prices: prices, recipes: recipes }; } function getExistingIds(sheet) { var data = sheet.getDataRange().getValues(); var ids = []; for (var i = 1; i < data.length; i++) { if (data[i][0]) ids.push(String(data[i][0])); } return ids; } function findRowById(sheet, id) { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id) || String(data[i][1]) === String(id)) { return i + 1; } } return -1; } function jsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
